										Previously we built an edge incidence-style repersentation; we will now repersent a Markov transition matrix.
										NOTE // S=[1,2,3,4,5,7,9] //
										
											let states=[1,2,3,4,5,7,9];

											let transitions = {
												3:[4,5,7,9,1,2,3],
												1:[4,5,7,9,1,2,3],
												7:[4,5,7,9,1,2,3]
												};

											let matrix = states.map(i => {
												let destinations = transitions[i] || [];

												return states.map(j => {
													return destinations.filter(x => x === j).length /
														destinations.length || 0;
												});
											});

											console.table(matrix);
